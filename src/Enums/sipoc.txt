<?php

declare(strict_types=1);

namespace Rimba\Work\Enums;

enum ActivityType: string
{
    case Create = 'create';
    case Capture = 'capture';
    case Verify = 'verify';
    case Analyze = 'analyze';
    case Decide = 'decide';
    case Authorize = 'authorize';
    case Transform = 'transform';
    case Execute = 'execute';
    case Record = 'record';
    case Communicate = 'communicate';

    public function label(): string
    {
        return str($this->value)->headline()->toString();
    }
}

enum ExecutionType: string
{
    case Human = 'human';
    case Rimba = 'rimba';
    case EventDriven = 'event_driven';

    public function createsInboxTask(): bool
    {
        return $this === self::Human;
    }

    public function executesImmediately(): bool
    {
        return $this === self::Rimba;
    }
}

enum WorkflowStatus: string
{
    case Draft = 'draft';
    case Active = 'active';
    case Waiting = 'waiting';
    case Completed = 'completed';
    case Cancelled = 'cancelled';
    case Failed = 'failed';
}

enum TaskStatus: string
{
    case Pending = 'pending';
    case Ready = 'ready';
    case Assigned = 'assigned';
    case Started = 'started';
    case Waiting = 'waiting';
    case Completed = 'completed';
    case Cancelled = 'cancelled';
    case Failed = 'failed';
}

enum ArtifactType: string
{
    case Data = 'data';
    case Record = 'record';
    case Document = 'document';
    case Material = 'material';
    case Decision = 'decision';
    case Notification = 'notification';
}
